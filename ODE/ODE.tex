	\chapter{微分方程的基本概念}
	\section{微分方程的定义}
		
		\begin{definition}[微分方程]
		
			微分方程是联系着自变量和未知函数以及未知函数导数的方程。
		
		\end{definition}
		
		\begin{definition}[常微分方程]
			
			称如下形式的方程为常微分方程：$\displaystyle F(x,y,y',...,y^{(n)})=0,n\geq 1$
			，其中$F$为已知函数，自变量$x$在实数集$\mathbb{R}$的某个区间上变化，未知函数$y\in \mathbb{R}$，正整数$n$称作该常微分方程的阶。
			
		\end{definition}
		
		\begin{example}[以下方程不是常微分方程]
			
			$y'=f(y(x-1))\qquad y'=y(y(x))$
			
		\end{example}
		
		\begin{definition}[线性常微分方程]
			
			在\nameref{definition:常微分方程}中，若$F$关于$x,y',..,y^{(n)}$是线性的，则称该方程为线性常微分方程，简称为线性微分方程。
			
		\end{definition}
		
		\begin{definition}[方程的解]
			
			设$J$为$\mathbb{R}$中的区间，称函数$y=\Phi (x)(x\in J)$为方程$\displaystyle F(x,y,y',...,y^{(n)})=0,n\geq 1$的解，若
			$F(x,\Phi (x),\Phi '(x),..,\Phi ^{(n)})\equiv ,x\in J$，此时，称$J$为解的存在区间。
			
			若微分方程的解不包含任意常数，则称它为微分方程的一个特解。
			
			假设$y=\Phi (x;c_1,c_2,...,c_n)$是上述方程的解，其中$c_1,c_2,..,c_n$为任意常数，若它们相互独立，即jacobi行列式
			$\displaystyle \det{\frac{\partial (\Phi,\Phi ',...,\Phi ^{(n-1)})}{\partial (c_1,c_2,...,c_n)}\neq 0,x\in J}$
			,则称其为上述方程的通解。
			
			由隐式方程给出的通解称通积分，由隐式方程给出的特解称特积分。
			
		\end{definition}
		
		\begin{definition}[初值问题]
			
			设$y=\Phi (x)$是方程$\displaystyle F(x,y,y',...,y^{(n)})=0,n\geq 1$的解，且满足
			$\Phi (x_0)=y_0,...,\Phi ^{(n-1)}(x_0)=y_0^{(n-1)}$
			，称其为初始条件，并称$y=\Phi (x)$为上述方程满足上述初始条件的解。
			
			称此类问题为初值问题，称$y=\Phi (x)$为该初值问题的解。
			
		\end{definition}

	\section{微分方程的几何解释}

		\begin{definition}[积分曲线]
			考虑一阶微分方程$\displaystyle \frac{dy}{dx}=f(x,y)$，其中$f$
			在平面区域$G$上连续，设$y=\Phi (x),x\in J$是该方程的一个解，$J\subset \mathbb{R}$是区间，则平面点集
			$\Gamma =\{(x,y)|y=\Phi (x),x\in J \}$为平面上一条可微曲线，称这条曲线为解曲线或积分曲线。
		\end{definition}
		
		\begin{definition}[线素场]
			在区域$G$中的每一个点$P$处作一段以$f(P)$为斜率的小线段$l(P)$，称$l(P)$为方程$\displaystyle \frac{dy}{dx}=f(x,y)$
			在点$P$处的线素。称区域$G$连同它上面的全体线素为上述方程的线素场或方向场。
		\end{definition}
		
		\begin{example}[建立给定曲线族的积分方程]
			\textbf{I.}$y=(x-C)^3$
			
			需要注意求解此类问题需要使获得的方程的阶数尽可能低。
			
			由$y'=3(x-C)^2$可得$y'=3y^{\frac{2}{3}}$
			
			但由$y''=6(x-C)$可得$y=(\frac{y''}{6})^3$
			
			两个方程都有解$y=(x-C)^3$，但$\displaystyle y=(\frac{y''}{6})^3$有通解$\displaystyle \int \frac{dy}{\sqrt{9y^{\frac{4}{3}}+C_2}}=\pm x+C_3$
			,当$C_2=0$时才有$y=(x-C)^3$
			
			多余的阶数会使解空间的大小爆增。在一些特殊的方程中，阶数和解的常数数量相等，我们之后证明它（如果这里没有链接，提醒我补一下）
			
			\textbf{II.}$y=sin(x+C)$
			
			由$y'=cos(x+C)$可得$y^2+y'^2=1$，它的通解为上述方程，特解为$y\equiv 1 \text{和} y\equiv -1$
			
			由$y''=-sin(x+C)$可得$y''=-y$，它的通解为$y=C_1cosx+C_2sinx$，这再次说明上面的结论。
			
		\end{example}
