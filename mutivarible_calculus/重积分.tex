	\chapter{多元函数的极限和连续}
	\section{多元函数的极限}
		
		\begin{definition}[重极限]
		
			设$z=f(P)=f(x,y)$在$E\subset \mathbb{R}^2$上定义，$A$是一个定常数，$P_0=(x_0,y_0)$是$E$的极限点，若$\forall \epsilon >0,\exists \delta >0,s.t.$ $\displaystyle |f(P)-A|=|f(x,y)-A|<\epsilon $
			,其中
			$ \displaystyle (x,y)\in E \text{且} 0<\rho =d(P,P_0)<\delta$
			,则称当$P=(x,y)\to P_0=()x_0,y_0$时，函数$f(x,y)$存在重极限，极限值为$A$，记作
			$\displaystyle\lim\limits_{(x,y)\to (x_0,y_0)}f(x,y)=\lim\limits_{\substack{x\to x_0\\ y \to y_0}}f(x,y)=A$
		
		\end{definition}
		
		\begin{example}[\nameref{definition:重极限} 沿不同曲线逼近一致]
			
			考虑当$x\to o,y \to 0$时函数$\displaystyle\frac{xy}{x^2+y^2}$的极限。
			
			考虑动点$(x,y)$沿直线$y=kx$趋于$(0,0)$时，$\displaystyle \left.\frac{xy}{x^2+y^2}\right |_{y=kx}=\frac{kx^2}{(1+kk^2)x^2}=\frac{k}{1+k^2}$.
			
			与k有关，于是重极限不存在。
			
		\end{example}
		
		\begin{example}[\nameref{definition:重极限} 极坐标上关于\theta 一致]
			
			考虑重极限
			$\displaystyle I=\lim\limits_{\substack{x\to 0\\y\to 0}}x^2y^2ln(x^2+y^2)$,令$x=rcos\theta,y=rsin\theta$
			,于是
			$\displaystyle I=\lim\limits_{\substack{x\to 0\\ y \to 0}}x^2y^2ln(x^2+y^2)=\lim\limits_{(r,\theta)\to (0,0)}r^4cos^2\theta sin^2\theta\lim\limits_{(r,\theta)\to (0,0)}ln(r^2)\leq 2r^4ln(r)=0$
			
			
		\end{example}
		
		\begin{thmpf}[极限值的唯一性]
			
		设 $u_{0}$ 是 $E$ 的聚点，$f:E\to\mathbb R^{m}$ 在 $u_{0}$ 处分别以 $v_{0}$ 与 $v_{1}$ 为极限。则 $v_{0}=v_{1}$
		
		\pfp
		
		任给 $\varepsilon>0$。存在 $\delta_{0},\delta_{1}>0$，使 $0<\|u-u_{0}\|<\delta_{0}$ 时 $\|f(u)-v_{0}\|<\varepsilon$，$0<\|u-u_{0}\|<\delta_{1}$ 时 $\|f(u)-v_{1}\|<\varepsilon$。令 $\delta=\min\{\delta_{0},\delta_{1}\}$。由 $u_{0}$ 是 $E$ 的聚点，$\mathring B(u_{0},\delta)\cap E\neq\varnothing$，取其中一点 $u$，则
		$\displaystyle\|v_{0}-v_{1}\|\leqslant\|v_{0}-f(u)\|+\|f(u)-v_{1}\|<2\varepsilon$ 
		由 $\varepsilon$ 的任意性得 $\|v_{0}-v_{1}\|=0$，即 $v_{0}=v_{1}$
		
		\end{thmpf}
		
		\begin{thmpf}[极限的局部有界性]
			
			若 $\displaystyle\lim\limits_{u\to u_{0}}f(u)=v_{0}$，则存在 $\delta>0$ 使 $f\bigl(\mathring B(u_{0},\delta)\cap E\bigr)$ 有界，且 $\|f(u)\|\leqslant\|v_{0}\|+1$ 在此集合上成立
			
			\pfp
			
			在极限定义中取 $\varepsilon=1$，得 $\delta>0$ 使 $0<\|u-u_{0}\|<\delta$ 时 $\|f(u)-v_{0}\|<1$，于是
			$\|f(u)\|\leqslant\|f(u)-v_{0}\|+\|v_{0}\|<\|v_{0}\|+1$
			
		\end{thmpf}
		
		\begin{thmpf}[极限的四则运算]
			
			设 $\displaystyle f,g:E\to\mathbb R$，$\lim\limits_{u\to u_{0}}f(u)=a$，$\lim\limits_{u\to u_{0}}g(u)=b$。则
			\begin{enumerate}
				\item 对任意 $\lambda,\mu\in\mathbb R$，$\lim_{u\to u_{0}}\bigl(\lambda f(u)+\mu g(u)\bigr)=\lambda a+\mu b$；
				\item $\lim_{u\to u_{0}}f(u)g(u)=ab$；
				\item 若 $b\neq0$，则存在 $\delta>0$ 使 $u\in\mathring B(u_{0},\delta)\cap E$ 时 $g(u)\neq0$，且 $\lim_{u\to u_{0}}\frac{f(u)}{g(u)}=\frac{a}{b}$。
			\end{enumerate}
			
			\pfp
			
			(1) 若 $\lambda=\mu=0$，两边都是 $0$。否则令 $M=|\lambda|+|\mu|>0$。任给 $\varepsilon>0$，取 $\delta>0$ 使 $0<\|u-u_{0}\|<\delta$ 时
			$\displaystyle
			|f(u)-a|<\frac{\varepsilon}{M}, |g(u)-b|<\frac{\varepsilon}{M},
			$
			则
			$\displaystyle
			|\lambda f(u)+\mu g(u)-(\lambda a+\mu b)|
			\leqslant|\lambda|\,|f(u)-a|+|\mu|\,|g(u)-b|<\varepsilon .
			$
			
			(2) 由局部有界性，存在 $\delta_{1}>0$ 与 $M>0$ 使 $0<\|u-u_{0}\|<\delta_{1}$ 时 $|f(u)|\leqslant M$。任给 $\varepsilon>0$，取 $\delta\in(0,\delta_{1}]$ 使 $0<\|u-u_{0}\|<\delta$ 时
			$\displaystyle
			|g(u)-b|<\frac{\varepsilon}{2M}, |f(u)-a|<\frac{\varepsilon}{2(|b|+1)},
			$
			则
			$\displaystyle
			|f(u)g(u)-ab|\leqslant|f(u)|\,|g(u)-b|+|b|\,|f(u)-a|
			<\frac{\varepsilon}{2}+\frac{|b|\,\varepsilon}{2(|b|+1)}<\varepsilon .
			$
			
			(3) 取 $\delta_{1}>0$ 使 $0<\|u-u_{0}\|<\delta_{1}$ 时 $|g(u)-b|<\frac{|b|}{2}$，则 $|g(u)|>\frac{|b|}{2}\neq0$。任给 $\varepsilon>0$，取 $\delta\in(0,\delta_{1}]$ 使 $0<\|u-u_{0}\|<\delta$ 时
			$\displaystyle
			|f(u)-a|<\frac{|b|\,\varepsilon}{4},\qquad |g(u)-b|<\frac{|b|^{2}\varepsilon}{4(|a|+1)},
			$
			于是
			$\displaystyle
			\left|\frac{f(u)}{g(u)}-\frac{a}{b}\right|
			=\frac{|bf(u)-ag(u)|}{|g(u)|\,|b|}
			\leqslant\frac{2}{|b|^{2}}\Bigl(|b|\,|f(u)-a|+|a|\,|g(u)-b|\Bigr)
			<\frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\varepsilon .
			$
			
		\end{thmpf}
		
		极限值的唯一性，局部有界性，四则运算的分别进行性，上下极限的运算性质，保序性，极限收敛原理
		
		
		
		
		
		
		
		
			
	
	\chapter{多元函数微分学}
	\section{偏导数与全微分}
		
		\begin{definition}[偏导数]
			
			设$f(x,y)$在$(x_0,y_0)$点邻域内有定义，若极限
			$\displaystyle \lim\limits_{\Delta x \to 0} \frac{f(x_0+\Delta x,y_0)-f(x_0,y_0)}{\Delta x}$
			存在，则称其为$f(x,y)$在点$(x_0,y_0)$处关于$x$的\textbf{偏导数}，记作
			$\displaystyle \frac{\partial f(x_0,y_0)}{\partial x}\text{或}\left.\frac{\partial f}{\partial x}\right |_{(x_0,y_0)}\text{或}f_x'(x_0,y_0)$
			或 
			$f_1'(x_0,y_0)$
			而
			$\displaystyle \frac{\partial f(x,y)}{\partial x}$
			 、
			$\displaystyle \frac{\partial f}{\partial x}$
			 、
			$\displaystyle f_x'(x,y)\text{、} f_1'(x,y)$
			是$D$上的二元函数，简记作
			$\displaystyle \frac{\partial f}{\partial x}$
			或
			$\displaystyle f_x'$
			
		\end{definition}
			
			\begin{example}[\nameref{definition:偏导数}的一般例子]
				
				$u=x^ysinz(x>0)$的一阶偏导数:
				
				\[\frac{\partial u}{\partial x}=yx^{y-1}sinz \qquad \frac{\partial u}{\partial y}=x^ylnxsinz \qquad \frac{\partial u}{\partial z}=x^ycosz\]
				
			\end{example}
			
			\begin{example}[\nameref{definition:偏导数}只说明沿坐标轴方向连续]
				
				$f(x,y)=\begin{dcases}
						1,  &xy\neq 0\\
						0,  &xy=0
					\end{dcases}$
				
				有$f_x'(0,0)=f_y'(0,0)=0$，但函数在$(0,0)$不连续
				
			\end{example}
